cmake_minimum_required(VERSION 3.5)
project(g1_wbc_observer)
if("$ENV{ROS_VERSION}" STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  find_package(rviz_common REQUIRED)
  find_package(pluginlib REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(Qt5 REQUIRED COMPONENTS Widgets)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(QWT REQUIRED IMPORTED_TARGET Qt5Qwt6)
  add_library(g1_live_plots SHARED src/plot_panel.cpp)
  target_compile_features(g1_live_plots PRIVATE cxx_std_14)
  target_link_libraries(g1_live_plots Qt5::Widgets PkgConfig::QWT)
  ament_target_dependencies(g1_live_plots rviz_common pluginlib rclcpp sensor_msgs geometry_msgs)
  pluginlib_export_plugin_description_file(rviz_common plot_plugin.xml)
  install(TARGETS g1_live_plots LIBRARY DESTINATION lib)
  install(PROGRAMS scripts/observer.py scripts/hardware_observer.py DESTINATION lib/${PROJECT_NAME})
  install(FILES observer_foxy.launch.py walking_foxy.rviz DESTINATION share/${PROJECT_NAME})
  install(DIRECTORY ../g1_description/meshes DESTINATION share/${PROJECT_NAME}/g1_description)
  install(FILES ../g1_description/g1_29dof_mode_13.urdf DESTINATION share/${PROJECT_NAME}/g1_description)
  ament_package()
else()
  find_package(catkin REQUIRED)
  catkin_package()
endif()
